function renderChart(fileData, subjects) { const ctx = document.getElementById('comparisonChart').getContext('2d'); const isPercentage = document.getElementById('formatToggle').checked; if (chartInstance) { chartInstance.destroy(); } const datasets = coreYears.map(year => { return { label: year, data: subjects.map(subj => { const rate = fileData[subj][year]; if (rate === undefined) return 0; return isPercentage ? parseFloat((rate * 100).toFixed(2)) : parseFloat(rate.toFixed(4)); }), backgroundColor: yearColorMap[year], borderColor: yearColorMap[year], borderWidth: 1, barPercentage: 0.85, categoryPercentage: 0.75, maxBarThickness: 75 }; }); chartInstance = new Chart(ctx, { type: 'bar', data: { labels: subjects, datasets: datasets }, options: { responsive: true, maintainAspectRatio: false, // إضافة مساحة بادئة من الأعلى لمنع قص الأرقام layout: { padding: { top: 35 } }, scales: { y: { beginAtZero: true, // زيادة الحد الأقصى بنسبة 15% لترك مسافة للرقم max: isPercentage ? 115 : 1.15, grid: { color: '#e2e8f0' }, ticks: { display: false } }, x: { grid: { display: false }, ticks: { font: { size: 14, weight: '700' } } } }, plugins: { legend: { position: 'bottom', labels: { font: { size: 14, weight: 'bold' }, usePointStyle: false } }, datalabels: { anchor: 'end', align: 'top', offset: 4, font: { weight: 'bold', size: 13 }, formatter: (value) => { if (value === 0) return ''; return isPercentage ? value.toFixed(2) + '%' : value; }, color: '#1e293b' } } } }); }
