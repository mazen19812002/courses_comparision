function renderChart(deptData, subjects, metric) { const ctx = document.getElementById('comparisonChart').getContext('2d'); const isPercentage = document.getElementById('formatToggle').checked; const useAttendantsBase = document.getElementById('denomToggle').checked; const showCounts = document.getElementById('showCountsToggle').checked && (metric === 'PASS_RATE'); if (chartInstance) { chartInstance.destroy(); } let maxValFound = 0; let maxCountFound = 0; let datasets = []; if (metric === 'ALL_GRADES') { const selectedYear = document.getElementById('allGradesYearSelect').value; datasets = GRADE_KEYS.map(gk => { const dataValues = subjects.map(subj => { const yearData = deptData[subj] ? deptData[subj][selectedYear] : null; const valObj = getFormattedValue(yearData, gk, isPercentage, useAttendantsBase); if (valObj.numeric > maxValFound) maxValFound = valObj.numeric; return valObj.numeric; }); return { label: gk, data: dataValues, backgroundColor: gradeColorMap[gk] || '#0284c7', borderColor: gradeColorMap[gk] || '#0284c7', borderWidth: 1, barPercentage: 0.9, categoryPercentage: 0.8 }; }); } else if (metric === 'PASS_RATE' && showCounts) { // ألوان مميزة لكل عام ومؤشر لضمان وضوح الأعمدة المتجاورة const yearStyleMap = { '2024': { reg: '#94a3b8', att: '#64748b', rate: '#1d6385' }, '2025': { reg: '#fdba74', att: '#fb923c', rate: '#ea652d' }, '2026': { reg: '#86efac', att: '#4ade80', rate: '#1e6b28' } }; coreYears.forEach(year => { const regData = []; const attData = []; const rateData = []; subjects.forEach(subj => { const yearData = deptData[subj] ? deptData[subj][year] : null; const reg = yearData ? (yearData['TOTAL_STUDENTS'] || 0) : 0; const att = yearData ? (yearData['ATTENDANTS_COUNT'] || 0) : 0; const valObj = getFormattedValue(yearData, 'PASS_RATE', isPercentage, useAttendantsBase); if (reg > maxCountFound) maxCountFound = reg; if (att > maxCountFound) maxCountFound = att; if (valObj.numeric > maxValFound) maxValFound = valObj.numeric; regData.push(reg); attData.push(att); rateData.push(valObj.numeric); }); // 1. عمود المقيدين (Col D) - عمود مستقل متجاور datasets.push({ label: `${year} - مقيد`, data: regData, backgroundColor: yearStyleMap[year].reg, borderColor: yearStyleMap[year].att, borderWidth: 1, yAxisID: 'yCounts', barPercentage: 0.95, categoryPercentage: 0.85 }); // 2. عمود الحاضرين (Col E) - عمود مستقل متجاور datasets.push({ label: `${year} - حاضر`, data: attData, backgroundColor: yearStyleMap[year].att, borderColor: yearStyleMap[year].rate, borderWidth: 1, yAxisID: 'yCounts', barPercentage: 0.95, categoryPercentage: 0.85 }); // 3. عمود نسبة النجاح (Col S) - عمود مستقل متجاور datasets.push({ label: `${year} - نسبة %`, data: rateData, backgroundColor: yearStyleMap[year].rate, borderColor: yearStyleMap[year].rate, borderWidth: 1, yAxisID: 'yRate', barPercentage: 0.95, categoryPercentage: 0.85 }); }); } else { datasets = coreYears.map(year => { const dataValues = subjects.map(subj => { const yearData = deptData[subj] ? deptData[subj][year] : null; const valObj = getFormattedValue(yearData, metric, isPercentage, useAttendantsBase); if (valObj.numeric > maxValFound) maxValFound = valObj.numeric; return valObj.numeric; }); return { label: year, data: dataValues, backgroundColor: yearColorMap[year], borderColor: yearColorMap[year], borderWidth: 1, barPercentage: 0.85, categoryPercentage: 0.75, maxBarThickness: 75 }; }); } let scalesConfig = {}; if (metric === 'PASS_RATE' && showCounts) { scalesConfig = { x: { grouped: true, // فرض التجاوز الأفقي ومنع التراكب grid: { display: false }, ticks: { font: { size: 12, weight: '700' }, maxRotation: 45, minRotation: 0 } }, yCounts: { type: 'linear', position: 'right', beginAtZero: true, max: Math.ceil(maxCountFound * 1.25) || 50, title: { display: true, text: 'أعداد الطلاب (مقيد / حاضر)', font: { weight: 'bold' } }, grid: { color: '#e2e8f0' } }, yRate: { type: 'linear', position: 'left', beginAtZero: true, max: isPercentage ? 115 : 1.15, title: { display: true, text: isPercentage ? 'نسبة النجاح (%)' : 'نسبة النجاح (عشري)', font: { weight: 'bold' } }, grid: { display: false } } }; } else { let yMax = 100; if (isPercentage) { yMax = maxValFound > 0 ? Math.ceil(maxValFound * 1.15) : 100; } else { if (metric === 'PASS_RATE') { yMax = 1.15; } else { yMax = maxValFound > 0 ? Math.ceil(maxValFound * 1.2) : 10; } } scalesConfig = { x: { grouped: true, grid: { display: false }, ticks: { font: { size: 13, weight: '700' }, maxRotation: 45, minRotation: 0 } }, y: { beginAtZero: true, max: yMax, grid: { color: '#e2e8f0' }, ticks: { display: false } } }; } chartInstance = new Chart(ctx, { type: 'bar', data: { labels: subjects, datasets: datasets }, options: { responsive: true, maintainAspectRatio: false, layout: { padding: { top: 35, bottom: 10 } }, scales: scalesConfig, plugins: { legend: { position: 'bottom', labels: { font: { size: 12, weight: 'bold' }, boxWidth: 16 } }, datalabels: { anchor: 'end', align: 'top', offset: 2, textAlign: 'center', font: { weight: 'bold', size: 9 }, formatter: (value, context) => { if (!value || value === 0) return ''; if (showCounts && context.dataset.yAxisID === 'yCounts') { return value; } if (isPercentage) { return value.toFixed(1) + '%'; } else { return value; } }, color: '#1e293b' } } } }); }
